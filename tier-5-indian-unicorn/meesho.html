Thread-safe Reserve + Fulfill
class InventoryService {
    Map<String, ReentrantLock> productLocks = new ConcurrentHashMap<>();
    Map<String, InventoryItem> inventory = new ConcurrentHashMap<>();

    boolean reserve(String productId, int qty) {
        ReentrantLock lock = productLocks.computeIfAbsent(productId, k -> new ReentrantLock());
        lock.lock();
        try {
            InventoryItem item = inventory.get(productId);
            if (item.availableQty < qty) return false;  // insufficient stock
            item.availableQty -= qty;
            item.reservedQty += qty;
            return true;
        } finally { lock.unlock(); }
    }

    void fulfill(String productId, int qty) { // on order shipped
        // lock, reservedQty -= qty, soldQty += qty
    }

    void release(String productId, int qty) { // on order cancel
        // lock, reservedQty -= qty, availableQty += qty
    }
}
