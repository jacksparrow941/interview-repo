3 Design a Sliding Window Rate Limiter (code-level)
rate-limitersliding-windowredis
Hard 90%
Sliding Window Log Algorithm
Store timestamps of each request in a sorted set. On each request: remove old entries, count remaining, allow if below limit.
In-Memory Thread-Safe
class SlidingWindowRateLimiter {
    private final int maxRequests;
    private final long windowMs;
    private final Map<String, Deque<Long>> windows = new ConcurrentHashMap<>();

    boolean allow(String clientId) {
        long now = System.currentTimeMillis();
        long cutoff = now - windowMs;

        Deque<Long> timestamps = windows.computeIfAbsent(clientId, k -> new ArrayDeque<>());

        synchronized (timestamps) {
            // Remove expired
            while (!timestamps.isEmpty() && timestamps.peekFirst() <= cutoff)
                timestamps.pollFirst();
            if (timestamps.size() >= maxRequests) return false;
            timestamps.addLast(now);
            return true;
        }
    }
}

6 Design a Rate Limiter (code-level)
token-bucketsliding-windowconcurrent
Hard 88%
Token Bucket Implementation
class TokenBucketRateLimiter {
    private final long capacity;
    private final double refillRatePerMs;
    private double tokens;
    private long lastRefillTime;

    public TokenBucketRateLimiter(long capacity, long refillPerSecond) {
        this.capacity = capacity;
        this.refillRatePerMs = refillPerSecond / 1000.0;
        this.tokens = capacity;
        this.lastRefillTime = System.currentTimeMillis();
    }

    public synchronized boolean allowRequest() {
        refill();
        if (tokens >= 1) { tokens--; return true; }
        return false;
    }

    private void refill() {
        long now = System.currentTimeMillis();
        double toAdd = (now - lastRefillTime) * refillRatePerMs;
        tokens = Math.min(capacity, tokens + toAdd);
        lastRefillTime = now;
    }
}
Per-Key Rate Limiter
class RateLimiterService {
    ConcurrentHashMap<String, TokenBucketRateLimiter> limiters = new ConcurrentHashMap<>();
    boolean allow(String key) {
        return limiters.computeIfAbsent(key,
            k -> new TokenBucketRateLimiter(100, 10)).allowRequest();
    }
}

7 Design a Snowflake-style Distributed ID Generator
snowflakedistributed-idepoch
Hard 88%
Twitter Snowflake Structure (64-bit)
| 1 bit sign | 41 bits timestamp (ms) | 10 bits machine ID | 12 bits sequence |

- Timestamp: ms since custom epoch (Jan 1, 2010)
  → supports 2^41 ms ≈ 69 years
- Machine ID: 10 bits → 1024 machines
- Sequence: 12 bits → 4096 IDs/ms/machine

Total: ~4 million IDs/second per machine, globally unique, sortable!
Implementation
class SnowflakeIdGenerator {
    private final long EPOCH = 1288834974657L;  // Nov 4, 2010
    private final long MACHINE_ID;              // 0-1023
    private long sequence = 0;
    private long lastTs = -1;

    synchronized long nextId() {
        long ts = System.currentTimeMillis();
        if (ts == lastTs) {
            sequence = (sequence + 1) & 0xFFF;  // 12-bit mask
            if (sequence == 0) ts = waitNextMs(lastTs);  // sequence exhausted
        } else { sequence = 0; }
        lastTs = ts;
        return ((ts - EPOCH) << 22) | (MACHINE_ID << 12) | sequence;
    }

    private long waitNextMs(long ts) {
        long now = System.currentTimeMillis();
        while (now <= ts) now = System.currentTimeMillis();
        return now;
    }
}

9 Design a Circuit Breaker
circuit-breakerstate-machineresilience
Hard 86%
States
CLOSED → calls go through. OPEN → calls fail-fast (no downstream). HALF_OPEN → probe call, if succeeds → CLOSED, else → OPEN.
Full Implementation
enum State { CLOSED, OPEN, HALF_OPEN }

class CircuitBreaker {
    State state = CLOSED;
    int failCount = 0, successCount = 0;
    final int failThreshold = 5;
    final int successThreshold = 2;
    Instant openedAt;
    final Duration timeout = Duration.ofSeconds(30);
    final ReentrantLock lock = new ReentrantLock();

    <T> T call(Supplier<T> fn) {
        lock.lock(); try {
            if (state == OPEN) {
                if (Duration.between(openedAt, Instant.now()).compareTo(timeout) > 0)
                    state = HALF_OPEN;
                else throw new CircuitOpenException();
            }
        } finally { lock.unlock(); }

        try {
            T result = fn.get();
            onSuccess(); return result;
        } catch (Exception e) { onFailure(); throw e; }
    }

    void onSuccess() {
        lock.lock(); try {
            if (state == HALF_OPEN && ++successCount >= successThreshold) {
                state = CLOSED; failCount = 0; successCount = 0;
            }
        } finally { lock.unlock(); }
    }

    void onFailure() {
        lock.lock(); try {
            if (++failCount >= failThreshold || state == HALF_OPEN) {
                state = OPEN; openedAt = Instant.now(); successCount = 0;
            }
        } finally { lock.unlock(); }
    }
}

class Job {
    String id; Runnable task; JobStatus status;
    long scheduledAt;  // epoch ms
    int priority, maxRetries, retryCount;
    String cronExpr;   // null for one-time jobs
}

class Scheduler {
    // Min-heap by scheduledAt
    PriorityQueue<Job> queue = new PriorityQueue<>(
        Comparator.comparingLong((Job j) -> j.scheduledAt)
                  .thenComparing(j -> -j.priority));
    ScheduledExecutorService executor = Executors.newScheduledThreadPool(4);
    Map<String, ScheduledFuture<?>> running = new ConcurrentHashMap<>();

    String schedule(Runnable task, long delayMs, int priority) {
        Job job = new Job(UUID.randomUUID().toString(), task, 
                          System.currentTimeMillis() + delayMs, priority);
        queue.offer(job);
        return job.id;
    }

    void processJobs() {
        while (!Thread.currentThread().isInterrupted()) {
            Job job = queue.peek();
            if (job == null || job.scheduledAt > System.currentTimeMillis()) {
                Thread.sleep(10); continue;
            }
            queue.poll();
            executor.submit(() -> executeWithRetry(job));
        }
    }

    void executeWithRetry(Job job) {
        try { job.task.run(); job.status = COMPLETED; }
        catch (Exception e) {
            if (job.retryCount++ < job.maxRetries) {
                job.scheduledAt = System.currentTimeMillis() + backoff(job.retryCount);
                queue.offer(job);
            } else { job.status = FAILED; }
        }
    }
}

class URLShortener {
    static final String CHARS = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ";
    AtomicLong counter = new AtomicLong(1);
    ConcurrentHashMap<String, URLEntry> store = new ConcurrentHashMap<>();

    String shorten(String url, int expiryDays) {
        long id = counter.getAndIncrement();
        String code = toBase62(id);
        store.put(code, new URLEntry(url, Instant.now().plusSeconds(86400L*expiryDays)));
        return "https://sho.rt/" + code;
    }

    String expand(String code) {
        URLEntry e = store.get(code);
        if (e == null || e.isExpired()) throw new NotFoundException();
        return e.originalUrl;
    }

    private String toBase62(long n) {
        StringBuilder sb = new StringBuilder();
        while (n > 0) { sb.insert(0, CHARS.charAt((int)(n%62))); n/=62; }
        return sb.toString();
    }
}

19 Design a Key-Value Store
lsm-treewalconcurrent
Hard 83%
Architecture
  • MemTable: In-memory sorted tree (TreeMap). Writes go here first.
  • WAL (Write-Ahead Log): Append-only log for crash recovery.
  • SSTable: Immutable sorted file flushed from MemTable when full.
  • Compaction: Background merge of SSTables (remove tombstones, deduplicate).
  • Bloom Filter: Per-SSTable to skip files that don't contain a key (fast miss detection).
Core Operations
class KVStore {
    TreeMap<String,String> memTable = new TreeMap<>();
    List<SSTable> ssTables = new ArrayList<>();
    WriteAheadLog wal;
    final int MEM_TABLE_LIMIT = 1_000_000;

    synchronized void put(String key, String value) {
        wal.append(key, value);
        memTable.put(key, value);
        if (memTable.size() >= MEM_TABLE_LIMIT) flush();
    }

    String get(String key) {
        if (memTable.containsKey(key)) return memTable.get(key);
        for (int i = ssTables.size()-1; i >= 0; i--) {
            if (ssTables.get(i).bloomFilter.mightContain(key))
                return ssTables.get(i).get(key);  // binary search in sorted file
        }
        return null;
    }
}

26 Design a Bloom Filter

What is a Bloom Filter
Space-efficient probabilistic data structure. Answers: 'definitely not in set' OR 'probably in set'. No false negatives, configurable false positive rate.
Implementation
class BloomFilter {
    private final BitSet bits;
    private final int numBits;
    private final int numHashFunctions;

    BloomFilter(int expectedItems, double falsePositiveRate) {
        // Optimal bit count: -n*ln(p) / (ln2)^2
        numBits = (int)(-expectedItems * Math.log(falsePositiveRate) / (Math.log(2)*Math.log(2)));
        // Optimal hash count: (m/n) * ln2
        numHashFunctions = (int)(numBits * 1.0 / expectedItems * Math.log(2));
        bits = new BitSet(numBits);
    }

    void add(String item) {
        for (int i = 0; i < numHashFunctions; i++)
            bits.set(Math.abs(hash(item, i)) % numBits);
    }

    boolean mightContain(String item) {
        for (int i = 0; i < numHashFunctions; i++)
            if (!bits.get(Math.abs(hash(item, i)) % numBits)) return false;  // Definitely not
        return true;  // Probably yes
    }

    private int hash(String s, int seed) {
        return Objects.hash(s, seed);
    }
}

Implementation
class ConnectionPool {
    BlockingDeque<Connection> pool;
    AtomicInteger totalCreated;
    int maxSize, minIdle;
    long acquireTimeoutMs;

    Connection borrow() throws Exception {
        Connection conn = pool.pollFirst(acquireTimeoutMs, MILLISECONDS);
        if (conn == null) {
            if (totalCreated.get() < maxSize) {
                conn = createConnection(); totalCreated.incrementAndGet();
            } else throw new TimeoutException("Pool exhausted");
        }
        if (!conn.isValid()) { conn.close(); return borrow(); }  // retry
        return conn;
    }

    void release(Connection conn) {
        if (conn.isValid()) pool.offerFirst(conn);
        else { conn.close(); totalCreated.decrementAndGet(); }
    }

    // Health check thread: periodically validate idle connections
    void startHealthCheck() {
        scheduler.scheduleAtFixedRate(() -> {
            pool.forEach(c -> { if (!c.isValid()) { pool.remove(c); totalCreated.decrementAndGet(); } });
            while (pool.size() < minIdle) pool.add(createConnection());
        }, 30, 30, SECONDS);
    }
}
