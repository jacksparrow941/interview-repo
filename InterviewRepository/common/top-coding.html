2 Number of Islands
dfsbfsunion-find
Medium 92%
Approach
DFS or BFS. When land cell found, increment count and DFS to mark all connected land as visited.
Solution
def numIslands(grid):
    if not grid: return 0
    count = 0
    def dfs(r, c):
        if r < 0 or r >= len(grid) or c < 0 or c >= len(grid[0]): return
        if grid[r][c] != '1': return
        grid[r][c] = '0'  # mark visited
        for dr, dc in [(0,1),(0,-1),(1,0),(-1,0)]:
            dfs(r+dr, c+dc)
    for r in range(len(grid)):
        for c in range(len(grid[0])):
            if grid[r][c] == '1':
                count += 1; dfs(r, c)
    return count
Union-Find alternative
find with path compression + union by rank. O(α(n)) per operation. Good for streaming updates.
Complexity
Time: O(m×n) | Space: O(m×n) recursion stack

5 LRU Cache — implement O(1) get/put using HashMap + DoublyLinkedList
lru-cachehashmapdoubly-linked-list
Hard 88%
Problem
Design a cache that evicts the Least Recently Used item when full. Both get() and put() must be O(1).
MUFG use case: Cache FX rates (capacity=1000, evict stale entries on overflow).
Design Diagram
  HashMap<key, Node>              DoublyLinkedList
  +----+--------+              HEAD <-> [A] <-> [B] <-> [C] <-> TAIL
  | A  | Node*A |              (MRU)                           (LRU)
  | B  | Node*B |
  | C  | Node*C |  get(A): move A to HEAD  -- O(1)
  +----+--------+  put(D): add D to HEAD, remove TAIL (C) -- O(1)
Java Implementation
class LRUCache {
    private final int capacity;
    private final Map<Integer, Node> map = new HashMap<>();
    private final Node head = new Node(0,0), tail = new Node(0,0);

    public LRUCache(int capacity) {
        this.capacity = capacity;
        head.next = tail; tail.prev = head;
    }

    public int get(int key) {
        if (!map.containsKey(key)) return -1;
        Node n = map.get(key);
        moveToFront(n);
        return n.val;
    }

    public void put(int key, int val) {
        if (map.containsKey(key)) {
            map.get(key).val = val;
            moveToFront(map.get(key));
        } else {
            if (map.size() == capacity)
                map.remove(removeLast());
            Node n = new Node(key, val);
            map.put(key, n);
            addToFront(n);
        }
    }

    private void addToFront(Node n) {
        n.next = head.next; n.prev = head;
        head.next.prev = n; head.next = n;
    }
    private void moveToFront(Node n) { remove(n); addToFront(n); }
    private void remove(Node n) { n.prev.next = n.next; n.next.prev = n.prev; }
    private int removeLast() {
        Node lru = tail.prev; remove(lru); return lru.key;
    }
    static class Node { int key, val; Node prev, next; Node(int k, int v){key=k;val=v;} }
}
// Java shortcut: LinkedHashMap(capacity, 0.75f, true) with removeEldestEntry

7 Three Sum
two-pointerssorting
Medium 85%
Approach
Sort array. For each element, use two pointers on the rest to find pairs summing to -nums[i]. Skip duplicates at each level.
Solution
def threeSum(nums):
    nums.sort(); res = []
    for i in range(len(nums)-2):
        if i > 0 and nums[i] == nums[i-1]: continue
        l, r = i+1, len(nums)-1
        while l < r:
            s = nums[i]+nums[l]+nums[r]
            if s == 0: res.append([nums[i],nums[l],nums[r]]); l+=1; r-=1
            while l < r and nums[l]==nums[l-1]: l+=1
            while l < r and nums[r]==nums[r+1]: r-=1
            elif s < 0: l+=1
            else: r-=1
    return res
Complexity
Time: O(n²) | Space: O(1) excluding output
Follow-ups
• How does 4-Sum extend this? (add outer loop, O(n³))
• What is the hash-map approach and why is two-pointer preferred?

8 Sliding Window — max sum subarray and longest substring without repeating chars
sliding-windowtwo-pointersubarray
Medium 85%
Pattern Overview
Sliding Window: Maintain a window [left, right]. Expand right, shrink left when condition violated. Avoids O(n²) nested loops → O(n).
MUFG use case: Max transaction volume in any 5-minute window for anomaly detection.
Fixed Window — Max Sum of K Elements
public int maxSum(int[] arr, int k) {
    int windowSum = 0, maxSum = 0;
    for (int i = 0; i < k; i++) windowSum += arr[i];  // first window
    maxSum = windowSum;
    for (int i = k; i < arr.length; i++) {
        windowSum += arr[i] - arr[i - k];  // slide: add right, remove left
        maxSum = Math.max(maxSum, windowSum);
    }
    return maxSum;
}
Variable Window — Longest Substring No Repeats (LC-3)
public int lengthOfLongestSubstring(String s) {
    Map<Character, Integer> freq = new HashMap<>();
    int left = 0, maxLen = 0;
    for (int right = 0; right < s.length(); right++) {
        freq.merge(s.charAt(right), 1, Integer::sum);
        while (freq.get(s.charAt(right)) > 1) {
            freq.merge(s.charAt(left), -1, Integer::sum);
            if (freq.get(s.charAt(left)) == 0) freq.remove(s.charAt(left));
            left++;
        }
        maxLen = Math.max(maxLen, right - left + 1);
    }
    return maxLen;
}

from collections import deque
def levelOrder(root):
    if not root: return []
    res, q = [], deque([root])
    while q:
        level = []
        for _ in range(len(q)):
            node = q.popleft()
            level.append(node.val)
            if node.left: q.append(node.left)
            if node.right: q.append(node.right)

    return res

def canFinish(numCourses, prerequisites):
    graph = [[] for _ in range(numCourses)]
    for a, b in prerequisites: graph[b].append(a)
    color = [0] * numCourses
    def dfs(v):
        if color[v] == 1: return False  # cycle
        if color[v] == 2: return True

17 Meeting Rooms II
heapgreedyintervals
Medium 82%
Approach
DP with memoization. dp[i][j] = LCS length of s1[i:] and s2[j:]. If chars match: 1 + dp[i+1][j+1]. Else max(dp[i+1][j], dp[i][j+1]).
Solution
def longestCommonSubsequence(text1, text2):
    m, n = len(text1), len(text2)
    dp = [[0]*(n+1) for _ in range(m+1)]
    for i in range(m-1,-1,-1):
        for j in range(n-1,-1,-1):
            if text1[i]==text2[j]:
                dp[i][j] = 1 + dp[i+1][j+1]
            else:
                dp[i][j] = max(dp[i+1][j], dp[i][j+1])
    return dp[0][0]
Complexity
Time: O(m×n) | Space: O(m×n), optimizable to O(min(m,n))

public int[] merge(int[] a, int[] b) {
    int i=0, j=0, k=0;
    int[] res = new int[a.length+b.length];
    while (i<a.length && j<b.length)
        res[k++] = a[i]<=b[j] ? a[i++] : b[j++];

def productExceptSelf(nums):
    n = len(nums); res = [1]*n
    # Left pass
    prefix = 1
    for i in range(n):

26 Trapping Rain Water
two-pointersstack
Hard 78%
Approach
Two pointer from both ends. Track left_max and right_max. Water at index i = min(left_max, right_max) - height[i]. Move the shorter side inward.
Solution
def trap(height):
    l, r = 0, len(height)-1
    lmax = rmax = water = 0
    while l < r:
        if height[l] <= height[r]:
            if height[l] >= lmax: lmax = height[l]
            else: water += lmax - height[l]
            l += 1
        else:
            if height[r] >= rmax: rmax = height[r]
            else: water += rmax - height[r]
            r -= 1
    return water

    while l < r:

    freq = {0: 1}
    for n in nums:
        prefix += n
        count += freq.get(prefix - k, 0)
        freq[prefix] = freq.get(prefix, 0) + 1
    return count

54 Network Delay Time (Dijkstra)
dijkstraheap
Medium 72%
Dijkstra's Algorithm
Min-heap (priority queue) for greedy shortest path. Relax all edges from current shortest node.
Solution
import heapq
def networkDelayTime(times, n, k):
    graph = defaultdict(list)
    for u,v,w in times: graph[u].append((v,w))
    dist = {k: 0}
    heap = [(0, k)]
    while heap:
        d, u = heapq.heappop(heap)
        if d > dist.get(u, float('inf')): continue
        for v, w in graph[u]:
            nd = d + w
            if nd < dist.get(v, float('inf')):
                dist[v] = nd; heapq.heappush(heap, (nd,v))
    return max(dist.values()) if len(dist)==n else -1
Complexity
Time: O((V+E) log V) | Space: O(V+E)
Follow-ups
• When does Dijkstra fail? (negative edge weights — use Bellman-Ford)
• All-pairs shortest path? (Floyd-Warshall O(V³))
• Negative cycle detection? (Bellman-Ford: after V-1 iterations, try once more)
